/*
 * Number of projects shown in one viewport (01 02 03 / 04 05 06 / 07 08 09 ...).
 */
const PROJECTS_PER_GROUP = 3;
const GROUP_COUNT = Math.ceil(projects.length / PROJECTS_PER_GROUP);
const LAST_GROUP_INDEX = GROUP_COUNT - 1;

const CARD_HEIGHT = 190;
const MOBILE_CARD_MIN_HEIGHT = 260;
const GAP = 14;
const CARD_SHADOW_OFFSET = 10; // the black offset card sticks out 10px below
// 3*190 + 2*14 + 10 = 608
const GROUP_HEIGHT =
  PROJECTS_PER_GROUP * CARD_HEIGHT +
  (PROJECTS_PER_GROUP - 1) * GAP +
  CARD_SHADOW_OFFSET;
const EXPANDED_HEIGHT = 598;

const GROUP_TRANSITION_MS = 650;
const TRANSITION_FAILSAFE_MS = GROUP_TRANSITION_MS + 200;
const PIN_SETTLE_MS = 250; // ignore input right after pinning
const EDGE_RELEASE_SETTLE_MS = GROUP_TRANSITION_MS + 250; // no release right after a group change
const WHEEL_GESTURE_IDLE_MS = 160;
const WHEEL_INPUT_GRACE_MS = 500;
const TOUCH_INPUT_GRACE_MS = 1200;
const JUMP_KEY_GRACE_MS = 1500;
const NATIVE_JUMP_THRESHOLD = 40;
const PIN_EPSILON = 1;
const DESKTOP_QUERY = "(min-width: 1024px)";

type Project = (typeof projects)[number];
type GallerySelection = { project: Project; imageIndex: number };
type ProjectsControllerState = "normal" | "pinned" | "transitioning" | "navbar";
type ScrollDirection = "down" | "up";
const projectImages = (project: Project) => project.images ?? [];

function scrollInstantlyTo(top: number) {
  const root = document.documentElement;
  const previousBehavior = root.style.scrollBehavior;

  root.style.scrollBehavior = "auto";
  window.scrollTo({ top, behavior: "auto" });
  root.style.scrollBehavior = previousBehavior;
}

/* True when the wheel/touch started inside the expanded card's own scroller AND it can still scroll that way. */
function innerScrollerCanScroll(
  target: EventTarget | null,
  direction: ScrollDirection,
) {
  if (!(target instanceof Element)) return false;
  const scroller = target.closest<HTMLElement>("[data-projects-scroll]");
  if (!scroller) return false;
  const max = scroller.scrollHeight - scroller.clientHeight;
  if (max <= 1) return false;
  return direction === "down"
    ? scroller.scrollTop < max - 1
    : scroller.scrollTop > 1;
}

export default function Projects() {
  const sectionRef = useRef<HTMLElement | null>(null);

  const activeGroupRef = useRef(0);
  const [activeGroup, setActiveGroup] = useState(0);
  const [isDesktop, setIsDesktop] = useState(true);
  const [controllerState, setControllerStateValue] =
    useState<ProjectsControllerState>("normal");
  const controllerStateRef = useRef<ProjectsControllerState>("normal");
  const transitionIdRef = useRef(0);
  const transitionTimerRef = useRef<number | null>(null);
  const lastScrollYRef = useRef<number | null>(null);
  const galleryOpenRef = useRef(false);
  const lastLockActivityRef = useRef(0);

  const [selectedProject, setSelectedProject] = useState<Project | null>(null);
  const [gallerySelection, setGallerySelection] =
    useState<GallerySelection | null>(null);

  const setControllerState = useCallback(
    (nextState: ProjectsControllerState) => {
      controllerStateRef.current = nextState;
      setControllerStateValue(nextState);
    },
    [],
  );

  const setGroup = useCallback((group: number) => {
    activeGroupRef.current = group;
    setActiveGroup(group);
  }, []);

  const clearTransitionTimer = useCallback(() => {
    if (transitionTimerRef.current !== null) {
      window.clearTimeout(transitionTimerRef.current);
      transitionTimerRef.current = null;
    }
  }, []);

  const finishGroupTransition = useCallback(
    (transitionId: number) => {
      if (
        transitionId !== transitionIdRef.current ||
        controllerStateRef.current !== "transitioning"
      )
        return;
      clearTransitionTimer();
      setControllerState("pinned");
    },
    [clearTransitionTimer, setControllerState],
  );

  const handleGroupTransitionEnd = (
    event: React.TransitionEvent<HTMLDivElement>,
  ) => {
    if (
      event.target !== event.currentTarget ||
      event.propertyName !== "transform"
    )
      return;
    finishGroupTransition(transitionIdRef.current);
  };

  const changeGroup = useCallback(
    (group: number) => {
      if (controllerStateRef.current !== "pinned") return;
      if (
        group < 0 ||
        group > LAST_GROUP_INDEX ||
        group === activeGroupRef.current
      )
        return;

      const transitionId = ++transitionIdRef.current;
      lastLockActivityRef.current = performance.now();
      setControllerState("transitioning");
      setGroup(group);

      // transitionend can be skipped (hidden tab, interrupted style change). Never stay locked.
      clearTransitionTimer();
      transitionTimerRef.current = window.setTimeout(
        () => finishGroupTransition(transitionId),
        TRANSITION_FAILSAFE_MS,
      );
    },
    [clearTransitionTimer, finishGroupTransition, setControllerState, setGroup],
  );

  const openProject = (project: Project) => setSelectedProject(project);
  const closeProject = () => setSelectedProject(null);

  /*
   * ==========================================================
   * LIGHTBOX SCROLL LOCK
   * ==========================================================
   */
  useEffect(() => {
    if (!gallerySelection) return;
    galleryOpenRef.current = true;

    const originalY = window.scrollY;
    const body = document.body;
    const root = document.documentElement;
    // Measure BEFORE hiding overflow; afterwards the scrollbar is gone and this reads 0.
    const scrollbarGap = window.innerWidth - root.clientWidth;
    const old = {
      bodyOverflow: body.style.overflow,
      rootOverflow: root.style.overflow,
      position: body.style.position,
      top: body.style.top,
      width: body.style.width,
      paddingRight: body.style.paddingRight,
    };

    body.style.overflow = "hidden";
    root.style.overflow = "hidden";
    body.style.position = "fixed";
    body.style.top = `-${originalY}px`;
    body.style.width = "100%";
    if (scrollbarGap > 0) body.style.paddingRight = `${scrollbarGap}px`;

    const handleKeyDown = (event: KeyboardEvent) => {
      if (event.key === "Escape") setGallerySelection(null);
      if (event.key === "ArrowLeft")
        setGallerySelection(
          (current) =>
            current && {
              ...current,
              imageIndex:
                (current.imageIndex -
                  1 +
                  projectImages(current.project).length) %
                projectImages(current.project).length,
            },
        );
      if (event.key === "ArrowRight")
        setGallerySelection(
          (current) =>
            current && {
              ...current,
              imageIndex:
                (current.imageIndex + 1) %
                projectImages(current.project).length,
            },
        );
    };
    window.addEventListener("keydown", handleKeyDown);

    return () => {
      window.removeEventListener("keydown", handleKeyDown);
      body.style.overflow = old.bodyOverflow;
      root.style.overflow = old.rootOverflow;
      body.style.position = old.position;
      body.style.top = old.top;
      body.style.width = old.width;
      body.style.paddingRight = old.paddingRight;
      galleryOpenRef.current = false;
      lastScrollYRef.current = null; // the restore below must not look like a scroll direction
      scrollInstantlyTo(originalY);
    };
  }, [gallerySelection]);

  /*
   * ==========================================================
   * PROJECTS SCROLL CONTROLLER
   * ==========================================================
   *
   * - Position decides everything; wheel deltas are only used to pre-empt.
   * - Entering from above  -> group 0.  Entering from below -> last group.
   * - While pinned/transitioning, input is cancelled and the document is
   *   re-aligned on every input.
   * - Release happens at the first/last group after the settle period.
   */
  useEffect(() => {
    const section = sectionRef.current;
    if (!section) return;

    const desktop = window.matchMedia(DESKTOP_QUERY);
    setIsDesktop(desktop.matches);

    let wheelIdleTimer: number | null = null;
    let wheelConsumed = false;
    let touchY: number | null = null;
    let touchConsumed = false;
    let inputGraceUntil = 0;
    let jumpUntil = 0;

    const isEnabled = () => desktop.matches && !galleryOpenRef.current;
    const hasRecentInput = () => performance.now() < inputGraceUntil;
    const isJumping = () => performance.now() < jumpUntil;
    const noteInput = (graceMs: number) => {
      inputGraceUntil = performance.now() + graceMs;
      jumpUntil = 0;
    };

    const armWheelIdleTimer = () => {
      if (wheelIdleTimer !== null) window.clearTimeout(wheelIdleTimer);
      wheelIdleTimer = window.setTimeout(() => {
        wheelConsumed = false;
        wheelIdleTimer = null;
      }, WHEEL_GESTURE_IDLE_MS);
    };

    // A gesture that pinned the section must not also advance a group.
    const consumeActiveGesture = () => {
      wheelConsumed = true;
      armWheelIdleTimer();
      if (touchY !== null) touchConsumed = true;
    };

    const navbarBusy = () => {
      if (controllerStateRef.current === "navbar" && !isNavbarNavigating()) {
        setControllerState("normal"); // END event was missed; don't stay locked out
      }
      return controllerStateRef.current === "navbar" || isNavbarNavigating();
    };

    // When Projects is fully off-screen, choose the group the next entry should show.
    const syncGroupToPosition = () => {
      const rect = section.getBoundingClientRect();
      if (rect.bottom <= 0) {
        if (activeGroupRef.current !== LAST_GROUP_INDEX)
          setGroup(LAST_GROUP_INDEX);
        return true;
      }
      if (rect.top >= window.innerHeight) {
        if (activeGroupRef.current !== 0) setGroup(0);
        return true;
      }
      return false;
    };

    const canPin = (direction: ScrollDirection, top: number, distance: number) => {
      const height = section.offsetHeight;
      return direction === "down"
        ? activeGroupRef.current < LAST_GROUP_INDEX &&
            top <= distance &&
            top > -height
        : // Up: pin as soon as the section's bottom edge is in (or about to enter)
          // the viewport, so the page never scrolls through Projects.
          activeGroupRef.current > 0 &&
            top <= PIN_EPSILON &&
            top > -height - distance;
    };

    const pinProjects = () => {
      if (controllerStateRef.current !== "normal") return;
      lastLockActivityRef.current = performance.now();
      setControllerState("pinned");
      scrollInstantlyTo(window.scrollY + section.getBoundingClientRect().top);
    };

    // Put the section exactly back on the viewport after layout shifts or stray scroll.
    const realign = () => {
      const top = section.getBoundingClientRect().top;
      if (Math.abs(top) > PIN_EPSILON) scrollInstantlyTo(window.scrollY + top);
    };

    const realignIfPinned = () => {
      if (!desktop.matches || galleryOpenRef.current) return;
      const state = controllerStateRef.current;
      if (state === "pinned" || state === "transitioning") realign();
    };

    const forceRelease = (group: number) => {
      clearTransitionTimer();
      transitionIdRef.current += 1;
      setGroup(group);
      setControllerState("normal");
    };

    const handleIntent = (
      direction: ScrollDirection,
      distance: number,
      preventDefault: () => void,
      isTouch = false,
    ) => {
      if (distance <= 0) return;
      const state = controllerStateRef.current;

      if (state === "pinned" || state === "transitioning") realign();

      if (state === "transitioning") {
        preventDefault();
        return;
      }

      if (state === "pinned") {
        if (isTouch && touchConsumed) {
          preventDefault();
          return;
        }
        // Swallow the tail of the gesture that pinned the section.
        const sinceActivity = performance.now() - lastLockActivityRef.current;
        if (sinceActivity < PIN_SETTLE_MS) {
          preventDefault();
          return;
        }
        const nextGroup =
          activeGroupRef.current + (direction === "down" ? 1 : -1);
        if (nextGroup < 0 || nextGroup > LAST_GROUP_INDEX) {
          // Edge: only release on a fresh gesture, not momentum from the last group change.
          if (sinceActivity < EDGE_RELEASE_SETTLE_MS) {
            preventDefault();
            return;
          }
          setControllerState("normal");
          return;
        }
        preventDefault();
        if (isTouch) touchConsumed = true;
        changeGroup(nextGroup);
        return;
      }

      if (
        state === "normal" &&
        canPin(direction, section.getBoundingClientRect().top, distance)
      ) {
        preventDefault();
        if (isTouch) touchConsumed = true;
        pinProjects();
      }
    };

    const handleWheel = (event: WheelEvent) => {
      if (event.ctrlKey || event.deltaY === 0) return; // ctrl+wheel = zoom
      if (!isEnabled() || navbarBusy()) return;
      noteInput(WHEEL_INPUT_GRACE_MS);

      const delta =
        event.deltaMode === WheelEvent.DOM_DELTA_LINE
          ? event.deltaY * 16
          : event.deltaMode === WheelEvent.DOM_DELTA_PAGE
            ? event.deltaY * window.innerHeight
            : event.deltaY;
      const direction: ScrollDirection = delta > 0 ? "down" : "up";

      // Let the expanded card scroll its own content without changing groups.
      if (innerScrollerCanScroll(event.target, direction)) {
        wheelConsumed = true;
        armWheelIdleTimer();
        return;
      }

      armWheelIdleTimer();
      const state = controllerStateRef.current;
      if (wheelConsumed && (state === "pinned" || state === "transitioning")) {
        event.preventDefault();
        return;
      }

      handleIntent(direction, Math.abs(delta), () => {
        wheelConsumed = true;
        event.preventDefault();
      });
    };

    const handleTouchStart = (event: TouchEvent) => {
      touchY = event.touches[0]?.clientY ?? null;
      touchConsumed = false;
    };

    const handleTouchMove = (event: TouchEvent) => {
      const nextY = event.touches[0]?.clientY;
      if (touchY === null || nextY === undefined) return;
      const movement = touchY - nextY;
      touchY = nextY;
      if (movement === 0) return;
      if (!isEnabled() || navbarBusy()) return;
      noteInput(TOUCH_INPUT_GRACE_MS);

      const direction: ScrollDirection = movement > 0 ? "down" : "up";
      if (innerScrollerCanScroll(event.target, direction)) {
        touchConsumed = true;
        return;
      }
      handleIntent(
        direction,
        Math.abs(movement),
        () => {
          if (event.cancelable) event.preventDefault();
        },
        true,
      );
    };

    const handleTouchEnd = () => {
      touchY = null;
      touchConsumed = false;
      noteInput(TOUCH_INPUT_GRACE_MS); // keep the grace window open for touch momentum
    };

    const handleKeyDown = (event: KeyboardEvent) => {
      if (
        event.defaultPrevented ||
        event.altKey ||
        event.ctrlKey ||
        event.metaKey
      )
        return;
      if (!isEnabled() || navbarBusy()) return;

      const target = event.target;
      if (target instanceof HTMLElement) {
        if (
          target.isContentEditable ||
          /^(INPUT|TEXTAREA|SELECT)$/.test(target.tagName)
        )
          return;
        if (event.key === " " && target.closest("button, a, summary")) return;
      }

      noteInput(WHEEL_INPUT_GRACE_MS);
      const state = controllerStateRef.current;

      if (event.key === "Home" || event.key === "End") {
        // Jump keys aren't a one-group gesture: leave at the matching edge and let the browser jump.
        if (state === "pinned" || state === "transitioning") {
          forceRelease(event.key === "End" ? LAST_GROUP_INDEX : 0);
        }
        jumpUntil = performance.now() + JUMP_KEY_GRACE_MS;
        return;
      }

      const direction: ScrollDirection | null =
        event.key === "ArrowDown" || event.key === "PageDown"
          ? "down"
          : event.key === "ArrowUp" || event.key === "PageUp"
            ? "up"
            : event.key === " "
              ? event.shiftKey
                ? "up"
                : "down"
              : null;
      if (!direction) return;

      // Holding a key must not chain through every group and out of the section.
      if (event.repeat && state !== "normal") {
        event.preventDefault();
        return;
      }

      handleIntent(
        direction,
        event.key === "ArrowDown" || event.key === "ArrowUp"
          ? 40
          : window.innerHeight,
        () => event.preventDefault(),
      );
    };

    /*
     * Safety net. Runs after the browser has moved the document, so it also
     * catches smooth-scroll animation, momentum and fast flicks that passed the
     * boundary between wheel events.
     */
    const handleScroll = () => {
      const y = window.scrollY;
      const previous = lastScrollYRef.current;
      lastScrollYRef.current = y;

      if (!isEnabled()) {
        lastScrollYRef.current = null;
        return;
      }
      if (navbarBusy()) return;

      const state = controllerStateRef.current;
      const rect = section.getBoundingClientRect();

      if (state === "pinned" || state === "transitioning") {
        if (Math.abs(rect.top) <= PIN_EPSILON) return;
        if (!hasRecentInput() && Math.abs(rect.top) > NATIVE_JUMP_THRESHOLD) {
          // Scrollbar drag / autoscroll / anchor jump: the visitor is leaving on purpose.
          forceRelease(rect.top < 0 ? LAST_GROUP_INDEX : 0);
          return;
        }
        scrollInstantlyTo(y + rect.top);
        return;
      }

      if (state !== "normal") return;
      if (syncGroupToPosition()) return;
      if (isJumping() || previous === null || previous === y) return;

      // Only pin movement that came from the visitor, not a scrollbar drag.
      if (!hasRecentInput()) return;

      const direction: ScrollDirection = y > previous ? "down" : "up";
      if (canPin(direction, rect.top, 0)) {
        consumeActiveGesture();
        pinProjects();
      }
    };

    const handleNavbarStart = () => {
      clearTransitionTimer();
      transitionIdRef.current += 1;
      wheelConsumed = false;
      touchConsumed = false;
      if (wheelIdleTimer !== null) {
        window.clearTimeout(wheelIdleTimer);
        wheelIdleTimer = null;
      }
      setControllerState("navbar");
    };

    const handleNavbarEnd = () => {
      if (controllerStateRef.current !== "navbar") return;
      setControllerState("normal");
      lastScrollYRef.current = window.scrollY;
      if (!desktop.matches) return;

      // The navbar landed on Projects itself: show the first group, pinned.
      if (
        Math.abs(section.getBoundingClientRect().top) <
        window.innerHeight * 0.5
      ) {
        setGroup(0);
        pinProjects();
      } else {
        syncGroupToPosition();
      }
    };

    const resizeObserver =
      typeof ResizeObserver !== "undefined"
        ? new ResizeObserver(realignIfPinned)
        : null;
    resizeObserver?.observe(document.body);

    const handleDesktopChange = () => {
      setIsDesktop(desktop.matches);
      if (!desktop.matches) {
        clearTransitionTimer();
        transitionIdRef.current += 1;
        setControllerState("normal");
        setGroup(0);
      }
    };

    window.addEventListener("wheel", handleWheel, {
      capture: true,
      passive: false,
    });
    window.addEventListener("touchstart", handleTouchStart, {
      capture: true,
      passive: true,
    });
    window.addEventListener("touchmove", handleTouchMove, {
      capture: true,
      passive: false,
    });
    window.addEventListener("touchend", handleTouchEnd, {
      capture: true,
      passive: true,
    });
    window.addEventListener("touchcancel", handleTouchEnd, {
      capture: true,
      passive: true,
    });
    window.addEventListener("keydown", handleKeyDown, true);
    window.addEventListener("scroll", handleScroll, { passive: true });
    window.addEventListener("resize", realignIfPinned);
    window.addEventListener(NAVBAR_NAVIGATION_START, handleNavbarStart);
    window.addEventListener(NAVBAR_NAVIGATION_END, handleNavbarEnd);
    desktop.addEventListener("change", handleDesktopChange);

    lastScrollYRef.current = window.scrollY;
    syncGroupToPosition();

    return () => {
      window.removeEventListener("wheel", handleWheel, true);
      window.removeEventListener("touchstart", handleTouchStart, true);
      window.removeEventListener("touchmove", handleTouchMove, true);
      window.removeEventListener("touchend", handleTouchEnd, true);
      window.removeEventListener("touchcancel", handleTouchEnd, true);
      window.removeEventListener("keydown", handleKeyDown, true);
      window.removeEventListener("scroll", handleScroll);
      window.removeEventListener("resize", realignIfPinned);
      resizeObserver?.disconnect();
      window.removeEventListener(NAVBAR_NAVIGATION_START, handleNavbarStart);
      window.removeEventListener(NAVBAR_NAVIGATION_END, handleNavbarEnd);
      desktop.removeEventListener("change", handleDesktopChange);
      if (wheelIdleTimer !== null) window.clearTimeout(wheelIdleTimer);
      clearTransitionTimer();
      transitionIdRef.current += 1;
      setControllerState("normal");
    };
  }, [changeGroup, clearTransitionTimer, setControllerState, setGroup]);

  return (
    <section
      ref={sectionRef}
      id="projects"
      className="
        relative
        z-20
        h-screen
        max-[1023px]:h-auto
        w-full
        bg-transparent
      "

                  selectedProject={selectedProject}
                  onOpen={openProject}
                  onClose={closeProject}

                    setGallerySelection({ project, imageIndex })
                  }
                />
              );

        </div>
      </div>
      {gallerySelection &&
        typeof document !== "undefined" &&
        createPortal(
          <ImageLightbox
            selection={gallerySelection}
            onClose={() => setGallerySelection(null)}
            onChange={(imageIndex) =>
              setGallerySelection(
                (current) => current && { ...current, imageIndex },
              )
            }
          />,
          document.body,
        )}
    </section>
  );
}

/*
 * =============================================================
 * PROJECT GROUP
 * =============================================================
 *
 * Each group is EXACTLY 608px:
 *
 * 190 + 14 + 190 + 14 + 190 = 608
 *
 * Therefore the viewport shows exactly 3 cards.
 */

function ProjectGroup({
  projects: groupProjects,
  selectedProject,
  onOpen,
  onClose,
  onOpenImage,
}: {
  projects: Project[];
  selectedProject: Project | null;
  onOpen: (project: Project) => void;
  onClose: () => void;
  onOpenImage: (project: Project, imageIndex: number) => void;
}) {
  const selectedIndex = selectedProject
    ? groupProjects.findIndex(
        (project) => project.number === selectedProject.number,
      )
    : -1;

  const isExpanded = selectedIndex !== -1;

  return (
    <div
      className="
        relative
        h-[608px]
        w-full
        shrink-0
        overflow-hidden
        max-[700px]:flex
        max-[700px]:flex-col
        max-[700px]:gap-[14px]
        max-[700px]:h-auto
        max-[700px]:overflow-visible
      "
    >
      {groupProjects.map((project, index) => {
        const isSelected = selectedProject?.number === project.number;

        /*
         * Normal positions:
         *
         * 01 → 0
         * 02 → 204
         * 03 → 408
         */

        const normalTop = index * (CARD_HEIGHT + GAP);
        const mobileCardHeight = isSelected
          ? EXPANDED_HEIGHT
          : MOBILE_CARD_MIN_HEIGHT;

        let translateY = 0;

        if (isExpanded) {
          /*
           * ---------------------------------------------------
           * SELECTED PROJECT
           * ---------------------------------------------------
           */

          if (isSelected) {
            translateY = -normalTop;
          } else if (index < selectedIndex) {
            /*
             * Projects above selected.
             */

            const selectedNormalTop = selectedIndex * (CARD_HEIGHT + GAP);

            translateY = -selectedNormalTop;
          } else {
            /*
             * Projects below selected.
             */

            const cardsBelow = index - selectedIndex - 1;

            const expandedStart = EXPANDED_HEIGHT + GAP;

            const additionalOffset = cardsBelow * (CARD_HEIGHT + GAP);

            const targetTop = expandedStart + additionalOffset;

            translateY = targetTop - normalTop;
          }
        }

        return (
          <div
            key={project.number}
            className="
                absolute
                left-0
                right-0
                transition-[transform,height]
                duration-[700ms]
                ease-[cubic-bezier(0.22,1,0.36,1)]
                max-[700px]:relative
                max-[700px]:inset-auto
                max-[700px]:w-full
                max-[700px]:shrink-0
                max-[700px]:transition-none
                max-[700px]:!h-auto
              "
            style={{
              top: normalTop,
              height: isSelected ? EXPANDED_HEIGHT : CARD_HEIGHT,
              ["--mobile-card-height" as string]: `${mobileCardHeight}px`,
              transform: `translate3d(0, ${translateY}px, 0)`,
              willChange: "transform, height",
            }}
          >
            <ProjectCard
              project={project}
              expanded={isSelected}
              onOpen={() => onOpen(project)}
              onClose={onClose}
              onOpenImage={(imageIndex) => onOpenImage(project, imageIndex)}
            />
          </div>
        );
      })}
    </div>
  );
}

function ImageLightbox({
  selection,
  onClose,
  onChange,
}: {
  selection: GallerySelection;
  onClose: () => void;
  onChange: (imageIndex: number) => void;
}) {
  const { project, imageIndex } = selection;
  const images = projectImages(project);
  return (
    <div
      className="fixed inset-0 z-[1000] flex items-center justify-center bg-black/85 p-4 sm:p-8"
      role="presentation"
      onMouseDown={(event) => {
        if (event.target === event.currentTarget) onClose();
      }}
    >
      <button
        type="button"
        onClick={onClose}
        aria-label="Close image"
        className="absolute right-4 top-4 z-10 rounded-full bg-white px-4 py-2 font-semibold text-[#111] shadow-lg sm:right-8 sm:top-8"
      >
        Close <span aria-hidden="true">×</span>
      </button>
      <div
        className="flex max-h-full max-w-full flex-col items-center gap-3"
        role="dialog"
        aria-modal="true"
        aria-label={`${project.title} screenshot ${imageIndex + 1}`}
      >
        <img
          src={images[imageIndex]}
          alt={`${project.title} screenshot ${imageIndex + 1}`}
          className="max-h-[82vh] max-w-[92vw] rounded-[8px] object-contain shadow-2xl"
          onMouseDown={(event) => event.stopPropagation()}
        />
        {images.length > 1 && (
          <div className="flex items-center gap-5 text-white">
            <button
              type="button"
              onClick={() =>
                onChange((imageIndex - 1 + images.length) % images.length)
              }
              aria-label="Previous screenshot"
              className="rounded px-3 py-1 hover:bg-white/15"
            >
              ← Previous
            </button>
            <span aria-live="polite">
              {imageIndex + 1} / {images.length}
            </span>
            <button
              type="button"
              onClick={() => onChange((imageIndex + 1) % images.length)}
              aria-label="Next screenshot"
              className="rounded px-3 py-1 hover:bg-white/15"
            >
              Next →
            </button>
          </div>
        )}
      </div>
    </div>
  );
}
